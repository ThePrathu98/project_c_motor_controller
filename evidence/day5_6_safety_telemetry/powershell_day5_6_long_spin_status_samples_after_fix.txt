PS C:\Users\pckal\esp_projects\project_c_motor_controller> function Send-MotorCommand {
    param([string]$cmd)

    $client = New-Object System.Net.Sockets.TcpClient($ip, $port)
    $stream = $client.GetStream()
    $writer = New-Object System.IO.StreamWriter($stream)
    $reader = New-Object System.IO.StreamReader($stream)

    $writer.AutoFlush = $true
    $writer.WriteLine($cmd)

    $response = $reader.ReadLine()

    $reader.Close()
    $writer.Close()
    $client.Close()

    Write-Host "$cmd -> $response"
    return $response
}

    $response = Send-MotorCommand "STATUS"

    $pattern = "state=(\S+) cmd=(-?\d+) target=(-?\d+) actual=(-?\d+) duty=(-?\d+) current_ma=(-?\d+) fault=(0x[0-9A-Fa-f]+) fault_name=(\S+) error=(-?\d+) delta=(-?\d+) missed=(\d+) step=(\d+)"

            missed      = [int]$matches[11]
            step        = [int]$matches[12]
            raw         = $response
        }
    }

    return [PSCustomObject]@{
        sample      = $Sample
        elapsed_s   = [math]::Round($Elapsed, 1)

    }
}
PS C:\Users\pckal\esp_projects\project_c_motor_controller> function Hold-Speed {
    param(
        [int]$Rpm,
        [int]$Seconds,
        [int]$SampleEverySeconds
    )

    Write-Host "`n--- Holding $Rpm RPM for $Seconds seconds ---"
    Send-MotorCommand "SET_SPEED $Rpm"

    $localStart = Get-Date
    while (((Get-Date) - $localStart).TotalSeconds -lt $Seconds) {
        Start-Sleep -Seconds $SampleEverySeconds

        $elapsed = ((Get-Date) - $global:TestStart).TotalSeconds
        $global:SampleNumber++

        $row = Read-MotorStatus -Sample $global:SampleNumber -Elapsed $elapsed -Phase "$Rpm RPM hold"
        $global:Rows += $row

        if ($row.fault_name -ne "NONE") {
            Write-Host "FAULT detected during $Rpm RPM hold. Stopping sample loop."
            break
        }
    }
}
